    seen: set = set()
    for ch in raw.ch_names:
        m = re.match(r"^(.*)-(\d+)$", ch)
        if m:
            base = m.group(1)
            if base not in seen:

                seen.add(base)
    if rename_map:

    missing = set(COMMON_CHANNELS) - set(raw.ch_names)
    if missing:
        return None

    raw.pick(COMMON_CHANNELS)
    raw.reorder_channels(COMMON_CHANNELS)
